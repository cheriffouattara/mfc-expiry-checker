⚙️ One-time Setup: Google Apps Script

To enable Push to Sheets, deploy a free Apps Script Web App (takes ~3 minutes).

  1. Open your Google Apps Script
  2. Create a New Project, paste the script below
  3. Click Deploy → New deployment → Web App
  4. Set Execute as: Me · Who has access: Anyone
  5. Copy the Web App URL and paste it below
var SHEET_ID = '12XbeGG7UBFtnX-GMw6HorhVJufm1AWz9y6R0iEA06iI'; function doPost(e) { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.openById(SHEET_ID); if (data.products) { var ws = ss.getSheetByName('Products') || ss.insertSheet('Products'); ws.clearContents(); var hdr = [['Level 1','Level 2','Level 3','Supplier Name','Product SKU', 'Product Name Local','Warehouses','VAT']]; ws.getRange(1,1,1,8).setValues(hdr); if (data.products.length > 0) ws.getRange(2,1,data.products.length,8).setValues(data.products); } if (data.expiry) { var we = ss.getSheetByName('Expiry') || ss.insertSheet('Expiry'); we.clearContents(); var hdrE = [['Level 1','Level 2','Level 3','Does it have expiry date?','Minimum shelf life']]; we.getRange(1,1,1,5).setValues(hdrE); if (data.expiry.length > 0) we.getRange(2,1,data.expiry.length,5).setValues(data.expiry); } return ContentService.createTextOutput( JSON.stringify({status:'ok'}) ).setMimeType(ContentService.MimeType.JSON); } function doGet(e) { return ContentService.createTextOutput( JSON.stringify({status:'ok',message:'Glovo Expiry Tool API ready'}) ).setMimeType(ContentService.MimeType.JSON); }
